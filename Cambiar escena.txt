using UnityEngine;
using UnityEngine.SceneManagement;

public class CambioDeEscena : MonoBehaviour
{
    public float tiempoParaCambio = 30f;

    {
        Invoke("CambiarEscena", tiempoParaCambio);
    }

    void CambiarEscena()
    {
        SceneManager.LoadScene("Winner");
    }

}
